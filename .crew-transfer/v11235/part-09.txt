function saveWorkerUpdates(updates){
  if(!state.service||state.service.closed||!updates.length)return;

  updates.forEach(update=>{
    if(!update||!update.workerId)return;

    const id=String(update.workerId);
    state.pendingWorkerUpdates[id]={
      ...(state.pendingWorkerUpdates[id]||{workerId:id}),
      ...update,
      workerId:id
    };
  });

  setSaveState('Guardando…','saving');
  clearTimeout(state.workerSaveTimer);
  state.workerSaveTimer=setTimeout(()=>flushPendingWorkerSaves(),220);
}

function flushPendingWorkerSaves(callback){
  if(typeof callback==='function'){
    state.workerSaveCallbacks.push(callback);
  }

  if(state.workerSaveInFlight)return;

  const batch=Object.values(state.pendingWorkerUpdates);

  if(!batch.length){
    runWorkerSaveCallbacks();
    return;
  }

  if(!state.service||state.service.closed){
    state.pendingWorkerUpdates={};
    runWorkerSaveCallbacks();
    return;
  }

  const serviceCode=state.service.code;
  state.pendingWorkerUpdates={};
  state.workerSaveInFlight=true;
  clearTimeout(state.workerSaveTimer);

  backend.run
    .withSuccessHandler(()=>{
      state.workerSaveInFlight=false;

      if(Object.keys(state.pendingWorkerUpdates).length){
        flushPendingWorkerSaves();
        return;
      }

      setSaveState('Guardado','saved');
      runWorkerSaveCallbacks();
    })
    .withFailureHandler(err=>{
      console.error(err);

      // Reincorporamos el lote para no perder cambios locales.
      batch.forEach(update=>{
        const id=String(update.workerId);
        state.pendingWorkerUpdates[id]={
          ...update,
          ...(state.pendingWorkerUpdates[id]||{})
        };
      });

      state.workerSaveInFlight=false;
      setSaveState('Error al guardar','error');
      state.workerSaveCallbacks=[];
    })
    .saveWorkerChanges(serviceCode,batch);
}

function runWorkerSaveCallbacks(){
  const callbacks=state.workerSaveCallbacks.splice(0);
  callbacks.forEach(fn=>{
    try{fn()}catch(err){console.error(err)}
  });
}

function nearestSpanishColorName(hex){
  const rgb=hexToRgb(hex);
  if(!rgb)return'COLOR';

  const palette=[
    ['NEGRO','#111111'],
    ['BLANCO','#FFFFFF'],
    ['GRIS','#9E9E9E'],
    ['ROJO','#E53935'],
    ['ROJO OSCURO','#8E0000'],
    ['NARANJA','#F58220'],
    ['AMARILLO','#FDD835'],
    ['LIMA','#8BC34A'],
    ['VERDE','#43A047'],
    ['VERDE OSCURO','#1B5E20'],
    ['TURQUESA','#00AFA0'],
    ['CELESTE','#29B6F6'],
    ['AZUL','#1E88E5'],
    ['AZUL OSCURO','#173F8A'],
    ['MORADO','#8E24AA'],
    ['VIOLETA','#673AB7'],
    ['ROSA','#D81B60'],
    ['FUCSIA','#E91E63'],
    ['MARRÓN','#795548'],
    ['BEIGE','#D7CCC8']
  ];

  let best='COLOR',bestDistance=Infinity;

  palette.forEach(([name,paletteHex])=>{
    const p=hexToRgb(paletteHex);
    const distance=
      Math.pow(rgb.r-p.r,2)+
      Math.pow(rgb.g-p.g,2)+
      Math.pow(rgb.b-p.b,2);

    if(distance<bestDistance){
      bestDistance=distance;
      best=name;
    }
  });

  return best;
}

function hexToRgb(hex){
  const m=String(hex||'').match(/^#([0-9a-fA-F]{2})([0-9a-fA-F]{2})([0-9a-fA-F]{2})$/);
  if(!m)return null;
  return{
    r:parseInt(m[1],16),
    g:parseInt(m[2],16),
    b:parseInt(m[3],16)
  };
}

async function parsePennyWisePdf(file){
  const data=await file.arrayBuffer();
  const pdf=await pdfjsLib.getDocument({data}).promise;

  const workers=[],blocks=[],blockById={};
  const meta={reference:'',serviceDate:'',venue:''};

  let currentBlock=null,currentServiceDate='',blockSeq=0;

  for(let pageNumber=1;pageNumber<=pdf.numPages;pageNumber++){
    const page=await pdf.getPage(pageNumber);
    const viewport=page.getViewport({scale:1});
    const width=viewport.width;
    const content=await page.getTextContent();

    const items=content.items.filter(i=>String(i.str||'').trim()).map(i=>({
      text:String(i.str||'').trim(),
      x:Number(i.transform[4]||0),
      y:Number(i.transform[5]||0),
      w:Number(i.width||0)
    }));

    const lines=groupItemsIntoLines(items);
    const pageLayout=detectPdfColumnLayout(lines,width);

    for(const lineItems of lines){
      const lineText=lineItems.slice().sort((a,b)=>a.x-b.x)
        .map(i=>i.text).join(' ').replace(/\s+/g,' ').trim();
      const upper=normalize(lineText);

      // Cada FECHA SERVICIO actualiza el día vigente para los bloques siguientes.
      if(upper.includes('FECHA SERVICIO:')){
        const match=lineText.match(/(\d{1,2}\/\d{1,2}\/\d{4})/);
        if(match){
          currentServiceDate=match[1];
          if(!meta.serviceDate)meta.serviceDate=currentServiceDate;
        }
      }

      if(pageNumber===1){
        if(!meta.reference&&upper.includes('REFERENCIA:')){
          meta.reference=extractBetween(lineText,'REFERENCIA:',['MÓVIL:','MOVIL:','CONTACTO:']);
        }
        if(!meta.venue&&upper.includes('RECINTO:')){
          meta.venue=extractBetween(lineText,'RECINTO:',['DIRECCIÓN:','DIRECCION:','TELÉFONO:','TELEFONO:']);
        }
      }

      if(upper.includes('TRABAJO:')){
        const rawWorkName=lineText.replace(/^.*?TRABAJO:\s*/i,'').trim();
        if(rawWorkName){
          const isReserve=/\+\s*$/.test(rawWorkName);
          const workName=rawWorkName.replace(/\+\s*$/,'').replace(/\s+/g,' ').trim();
          blockSeq++;
          currentBlock={
            id:'B'+blockSeq,
            trabajo:workName,
            rawTrabajo:rawWorkName,
            isReserve:isReserve,
            serviceDate:currentServiceDate||meta.serviceDate||'',
            firstStart:''
          };
          blocks.push(currentBlock);blockById[currentBlock.id]=currentBlock;
        }
        continue;
      }

      const rowNumberItem=findPdfRowNumber(lineItems,width,pageLayout);
      if(!rowNumberItem)continue;

      let personal=pageLayout
        ? columnTextByPdfLayout(lineItems,pageLayout,'personal')
        : columnText(lineItems,width,.06,.24);

      let especialidad=pageLayout
        ? columnTextByPdfLayout(lineItems,pageLayout,'especialidad')
        : columnText(lineItems,width,.24,.43);

      if(!currentBlock){
        blockSeq++;
        currentBlock={
          id:'B'+blockSeq,
          trabajo:'SIN BLOQUE',
          rawTrabajo:'SIN BLOQUE',
          isReserve:false,
          serviceDate:currentServiceDate||meta.serviceDate||'',
          firstStart:''
        };
        blocks.push(currentBlock);blockById[currentBlock.id]=currentBlock;
      }

      const shiftTimes=extractAllRowTimes(lineItems);

      let hLlam=pageLayout
        ? columnTextByPdfLayout(lineItems,pageLayout,'hLlam')
        : '';

      let inicio=pageLayout
        ? columnTextByPdfLayout(lineItems,pageLayout,'inicio')
        : '';

      let fin=pageLayout
        ? columnTextByPdfLayout(lineItems,pageLayout,'fin')
        : '';

      let pausaRaw=pageLayout
        ? columnTextByPdfLayout(lineItems,pageLayout,'pausa')
        : '';

      let dietaRaw=pageLayout
        ? columnTextByPdfLayout(lineItems,pageLayout,'dieta')
        : '';

      // Compatibilidad con la plantilla histórica Penny Wise.
      if(!hLlam)hLlam=shiftTimes[0] || columnText(lineItems,width,.425,.495);
      if(!inicio)inicio=shiftTimes[1] || columnText(lineItems,width,.475,.555);
      if(!fin)fin=shiftTimes[2] || columnText(lineItems,width,.525,.610);
      if(!pausaRaw)pausaRaw=shiftTimes[3] || '00:00';
      if(!dietaRaw)dietaRaw=shiftTimes[4] || '0';

      let csMarker=pageLayout
        ? columnTextByPdfLayout(lineItems,pageLayout,'cs')
        : columnText(lineItems,width,.680,.710);

      let telefono=pageLayout
        ? columnTextByPdfLayout(lineItems,pageLayout,'telefono')
        : columnText(lineItems,width,.765,.815);

      let movil=pageLayout
        ? columnTextByPdfLayout(lineItems,pageLayout,'movil')
        : columnText(lineItems,width,.815,.88);

      // Fallback para plantillas donde las columnas se han movido por completo.
      // Síntoma típico: PERSONAL contiene "07:45 08:00 14:00".
      if(looksLikeShiftBundle(personal)){
        const repaired=repairShiftedPdfWorkerRow(lineItems,{
          personal,
          especialidad,
          hLlam,
          inicio,
          fin,
          pausaRaw,
          dietaRaw,
          telefono,
          movil
        });

        personal=repaired.personal;
        especialidad=repaired.especialidad;
        hLlam=repaired.hLlam;
        inicio=repaired.inicio;
        fin=repaired.fin;
        pausaRaw=repaired.pausaRaw;
        dietaRaw=repaired.dietaRaw;
        telefono=repaired.telefono;
        movil=repaired.movil;
        csMarker=repaired.csMarker||csMarker;
      }

      if(!personal||['NOMBRE','PERSONAL'].includes(normalize(personal)))continue;

      const pausa=normalizePdfPauseValue(pausaRaw);
      const nDieta=normalizeDietFromPdf(dietaRaw);
      const isCoordinator=/\bX\b/i.test(csMarker)||/\bCS\b/i.test(especialidad);

      const rowNum=Number(rowNumberItem.text);
      const workerId=`D${dateId(currentBlock.serviceDate)}_P${pageNumber}_R${rowNum}_${currentBlock.id}`;

      const worker={
        id:workerId,page:pageNumber,pdfRow:rowNum,
        serviceDate:currentBlock.serviceDate||meta.serviceDate||'',
        blockId:currentBlock.id,trabajo:currentBlock.trabajo,
        isReserve:Boolean(currentBlock.isReserve),
        personal:cleanName(personal),especialidad:especialidad.trim(),
        hLlam:normalizeTime(hLlam),inicio:normalizeTime(inicio),fin:normalizeTime(fin),
        pausa:normalizeTime(pausa),
        nDieta:nDieta,
        isCoordinator:isCoordinator,
        movil:normalizePhone(movil||telefono)
      };

      workers.push(worker);
      if(!currentBlock.firstStart&&worker.inicio)currentBlock.firstStart=worker.inicio;
    }
  }

  blocks.forEach(b=>{
    b.label=(b.trabajo+(b.firstStart?' '+b.firstStart:'')).trim();
  });
  workers.forEach(w=>{
    const b=blockById[w.blockId];
    w.blockLabel=b?b.label:w.trabajo;
  });

  return {meta,workers};
}


function pdfHeaderKey(value){
  const v=normalize(String(value||''))
    .replace(/[.:]/g,' ')
    .replace(/\s+/g,' ')
    .trim();

  if(!v)return'';

  if(/^(N|NO|Nº|N°|NUM|NUMERO)$/.test(v))return'row';
  if(/\b(PERSONAL|NOMBRE|NOMBRES|TRABAJADOR)\b/.test(v))return'personal';
  if(/\b(ESPECIALIDAD|CATEGORIA|CATEGORIA PROFESIONAL)\b/.test(v))return'especialidad';
  if(/\bLLAM\b|\bLLAMADA\b/.test(v))return'hLlam';
  if(/^INICIO$|HORA INICIO/.test(v))return'inicio';
  if(/^FIN$|HORA FIN/.test(v))return'fin';
  if(/\bPAUSA\b/.test(v))return'pausa';
  if(/\bDIETA\b/.test(v))return'dieta';
  if(/^CS$|COORDINADOR/.test(v))return'cs';
  if(/\bMOVIL\b/.test(v))return'movil';
  if(/\bTELEFONO\b|^TEL$/.test(v))return'telefono';

  return'';
}

function detectPdfColumnLayout(lines,width){
  let best=null;

  for(const lineItems of lines){
    const items=lineItems.slice().sort((a,b)=>a.x-b.x);
    const found={};

    // PDF.js a veces separa "N. DIETA" o "H. LLAM." en varios items.
    for(let i=0;i<items.length;i++){
      for(let size=1;size<=3&&i+size<=items.length;size++){
        const chunk=items.slice(i,i+size);
        const text=chunk.map(x=>x.text).join(' ');
        const key=pdfHeaderKey(text);

        if(key&&!found[key]){
          found[key]={
            key,
            x:Math.min(...chunk.map(x=>x.x)),
            endX:Math.max(...chunk.map(x=>x.x+Number(x.w||0))),
            text
          };
        }
      }
    }

    const keys=Object.keys(found);
    const timeKeys=['hLlam','inicio','fin','pausa','dieta']
      .filter(k=>found[k]).length;

    const score=
      keys.length+
      (found.personal?3:0)+
      (found.especialidad?2:0)+
      (timeKeys>=2?3:0);

    if(
      score>=7 &&
      (found.personal||found.especialidad) &&
      timeKeys>=2 &&
      (!best||score>best.score)
    ){
      best={score,found};
    }
  }

  if(!best)return null;

  const columns=Object.values(best.found)
    .sort((a,b)=>a.x-b.x);

  const bounds={};

  columns.forEach((col,index)=>{
    const previous=columns[index-1];
    const next=columns[index+1];

    const min=previous
      ? (previous.x+col.x)/2
      : Math.max(0,col.x-width*.035);

    const max=next
      ? (col.x+next.x)/2
      : width+1;

    bounds[col.key]={min,max,x:col.x};
  });

  return {
    columns,
    bounds,
    keys:Object.keys(best.found)
  };
}

function columnTextByPdfLayout(items,layout,key){
  const bound=layout?.bounds?.[key];
  if(!bound)return'';

  return items
    .filter(i=>i.x>=bound.min&&i.x<bound.max)
    .sort((a,b)=>a.x-b.x)
    .map(i=>i.text)
    .join(' ')
    .replace(/\s+/g,' ')
    .trim();
}

function findPdfRowNumber(items,width,layout){
  const sorted=items.slice().sort((a,b)=>a.x-b.x);

  if(layout?.bounds?.row){
    const b=layout.bounds.row;
    const found=sorted.find(i=>
      i.x>=Math.max(0,b.min-width*.02) &&
      i.x<b.max &&
      /^\d{1,3}$/.test(String(i.text||'').trim())
    );
    if(found)return found;
  }

  // Las plantillas antiguas siempre llevan el nº de fila al extremo izquierdo.
  return sorted.find(i=>
    i.x<width*.065 &&
    /^\d{1,3}$/.test(String(i.text||'').trim())
  )||null;
}

function extractAllRowTimes(items){
  return items
    .slice()
    .sort((a,b)=>a.x-b.x)
    .map(i=>String(i.text||'').trim())
    .filter(text=>/^\d{1,2}[:.,]\d{2}$/.test(text))
    .map(normalizeTime);
}

function looksLikeShiftBundle(value){
  const times=String(value||'')
    .match(/\b\d{1,2}[:.,]\d{2}\b/g)||[];
  return times.length>=2;
}

function normalizePdfPauseValue(value){
  const raw=String(value||'').trim();

  if(!raw)return'00:00';
  if(/^(0|0[.,]0|0:00|00:00)$/.test(raw))return'00:00';

  const normalized=normalizeTime(raw);
  if(/^\d{2}:\d{2}$/.test(normalized))return normalized;

  const numeric=Number(raw.replace(',','.'));
  if(Number.isFinite(numeric)&&numeric>=0){
    const minutes=Math.round(numeric*60);
    return `${String(Math.floor(minutes/60)).padStart(2,'0')}:${String(minutes%60).padStart(2,'0')}`;
  }

  return'00:00';
}
