      if(Array.isArray(root.teams))state.service.teams=root.teams;
      if(Array.isArray(root.requirements))state.service.requirements=root.requirements;
      if(Array.isArray(root.audit))state.service.audit=root.audit;

      const nameEl=document.getElementById('serviceName');
      const dateEl=document.getElementById('serviceDate');
      const venueEl=document.getElementById('venue');

      if(nameEl&&document.activeElement!==nameEl)nameEl.value=state.service.serviceName||'';
      if(dateEl&&document.activeElement!==dateEl)dateEl.value=state.service.serviceDate||'';
      if(venueEl&&document.activeElement!==venueEl)venueEl.value=state.service.venue||'';

      const structureChanged=
        oldScenarios!==JSON.stringify(state.service.scenarios||[]) ||
        oldTeams!==JSON.stringify(state.service.teams||[]);

      const requirementsChanged=
        oldRequirements!==JSON.stringify(state.service.requirements||[]);

      const workPrefsChanged=
        oldWorkOrder!==JSON.stringify(state.service.workOrder||[]) ||
        oldHiddenWorkTypes!==JSON.stringify(state.service.hiddenWorkTypes||[]);

      if(structureChanged){
        renderScenarios();
        updateTeamSelects();
        renderPersonnel(true);
      }else if(workPrefsChanged){
        renderPersonnel(true);
        renderOperationalWorkFilters();
      }else if(requirementsChanged){
        // Only the staffing chips/summaries need refreshing.
        document.querySelectorAll('.work-group').forEach(group=>{
          const work=group.getAttribute('data-work-group');
          if(work)updateGroupSummary(work);
        });
        scheduleOperationalRefresh();
      }

      if(oldClosed!==state.service.closed)applyClosedState();
    },
    err=>console.error('[Firebase service listener]',firebaseErrorMessage(err))
  );

  const workersUnsub=fs.onSnapshot(
    fs.collection(db,'services',code,'workers'),
    snap=>{
      if(!state.service||String(state.service.code)!==code)return;

      if(firstWorkersSnapshot){
        firstWorkersSnapshot=false;
        // loadService() already populated the same collection.
        return;
      }

      snap.docChanges().forEach(change=>{
        const id=change.doc.id;

        if(change.type==='removed'){
          const previous=removeRealtimeWorker(id);
          if(previous){
            renderPersonnel(true);
            renderStats();
            scheduleOperationalRefresh();
          }
          return;
        }

        const worker=normalizeFirebaseWorker({id,...change.doc.data()});
        const previous=upsertRealtimeWorker(worker);

        // Ignore a listener echo if all visible/operational values are unchanged.
        if(previous && JSON.stringify(previous)===JSON.stringify(worker))return;

        refreshWorkerFromRealtime(worker,previous);
      });
    },
    err=>console.error('[Firebase workers listener]',firebaseErrorMessage(err))
  );

  const incidentsUnsub=fs.onSnapshot(
    fs.collection(db,'services',code,'incidents'),
    snap=>{
      if(!state.service||String(state.service.code)!==code)return;

      if(firstIncidentsSnapshot){
        firstIncidentsSnapshot=false;
        return;
      }

      let changed=false;
      const affectedWorkers=new Set();

      snap.docChanges().forEach(change=>{
        const id=change.doc.id;
        const rows=state.service.incidents||[];
        const idx=rows.findIndex(i=>String(i.id)===String(id));

        if(change.type==='removed'){
          if(idx>=0){
            affectedWorkers.add(String(rows[idx].workerId||''));
            rows.splice(idx,1);
            changed=true;
          }
          return;
        }

        const incident=normalizeFirebaseIncident({id,...change.doc.data()});
        if(idx>=0)rows[idx]=incident;
        else rows.push(incident);

        affectedWorkers.add(String(incident.workerId||''));
        changed=true;
      });

      if(changed){
        const unique=new Map();
        (state.service.incidents||[]).forEach(i=>{
          unique.set(String(i.id),i);
        });
        state.service.incidents=[...unique.values()];

        state.service.incidents.sort(
          (a,b)=>String(b.createdAt||'').localeCompare(String(a.createdAt||''))
        );
        renderIncidents();
        affectedWorkers.forEach(refreshIncidentWorkerBadge);
      }
    },
    err=>console.error('[Firebase incidents listener]',firebaseErrorMessage(err))
  );

  firebaseRealtimeUnsubs=[serviceUnsub,workersUnsub,incidentsUnsub];
}

async function firebaseSaveWorkerChanges(code,updates){
  const {db,fs}=await ensureFirebase();
  const ops=[];

  (updates||[]).forEach(update=>{
    if(!update||!update.workerId)return;
    const id=String(update.workerId);
    const local=state.service?.workers?.find(w=>String(w.id)===id);
    const patch={updatedAt:new Date().toISOString()};

    [
      'status','notes','included','isReserve','especialidad',
      'actualStart','actualEnd','actualPause','actualDiet',
      'ratingUniformidad','ratingActitud','ratingPuntualidad'
    ].forEach(key=>{
      if(Object.prototype.hasOwnProperty.call(update,key)){
        patch[key]=update[key];
      }
    });

    if(Object.prototype.hasOwnProperty.call(update,'teamId')){
      patch.teamId=String(update.teamId||'');
      const team=(state.service?.teams||[]).find(t=>t.id===patch.teamId);
      patch.scenarioId=team?String(team.scenarioId||''):'';
    }

    if(
      Object.prototype.hasOwnProperty.call(update,'status') &&
      update.status==='PRESENTE' &&
      local?.arrivalAt
    ){
      patch.arrivalAt=local.arrivalAt;
    }

    ops.push({
      type:'set',
      ref:fs.doc(db,'services',code,'workers',id),
      data:patch,
      merge:true
    });
  });

  await commitFirestoreOps(ops);

  return {ok:true,updated:ops.length};
}

async function firebaseSaveRequirements(code,serviceDate,trabajo,requirements,noTeams){

    ? [{

        requiredCount:0,
        updatedAt
      }]

async function firebaseSaveIncident(code,payload){
  const {db,fs}=await ensureFirebase();

  const editingId=String(payload.incidentId||'');
  const id=editingId||firebaseId('INC');
  const current=(state.service?.incidents||[]).find(i=>String(i.id)===id);
  const worker=(state.service?.workers||[]).find(
    w=>String(w.id)===String(payload.workerId||'')
  );

  const category=String(payload.category||'OTRO');
  const newCategory=String(payload.newCategory||'').trim();
  const oldCategory=category==='CAMBIO CATEGORIA'
    ? String(current?.oldCategory||worker?.especialidad||'')
    : '';

  const incident={
    id,
    workerId:worker?String(worker.id):'',
    personal:worker?String(worker.personal||''):'GENERAL',
    serviceDate:worker?String(worker.serviceDate||''):String(payload.serviceDate||''),
    trabajo:worker?String(worker.trabajo||''):String(payload.trabajo||'GENERAL'),
    createdAt:current?.createdAt||new Date().toISOString(),
    category,
    severity:String(payload.severity||'LEVE'),
    oldCategory,
    newCategory:category==='CAMBIO CATEGORIA'?newCategory:'',
    description:String(payload.description||'').trim(),
    actionTaken:String(payload.actionTaken||'').trim(),
    status:String(payload.status||'ABIERTA')
  };

  if(!incident.description)throw new Error('Describe la incidencia.');

  const batch=fs.writeBatch(db);
  batch.set(
    fs.doc(db,'services',code,'incidents',id),
    sanitizeFirestore(incident),
    {merge:true}
  );

  let returnedWorker=null;

  if(category==='CAMBIO CATEGORIA'){
    if(!worker)throw new Error('Selecciona un trabajador.');
    if(!newCategory)throw new Error('Indica la nueva categoría.');

    batch.set(
      fs.doc(db,'services',code,'workers',String(worker.id)),
      {especialidad:newCategory,updatedAt:new Date().toISOString()},
      {merge:true}
    );

    returnedWorker={...worker,especialidad:newCategory};
  }

  await batch.commit();


  return {ok:true,incident,worker:returnedWorker};
}

async function dispatchBackend(method,args){
  switch(String(method)){
    case 'loadService':
      return firebaseLoadService(...args);

    case 'generateNewService':
      return firebaseGenerateNewService();

    case 'duplicateServiceSetup':
      return firebaseDuplicateServiceSetup(...args);

    case 'saveServiceMeta':
      return firebaseSaveServiceMeta(...args);

    case 'saveScenariosAndTeams':
      return firebaseSaveStructure(...args);

    case 'addScenario':
      return firebaseAddScenario(...args);

    case 'saveWorkerChanges':
      return firebaseSaveWorkerChanges(...args);

    case 'saveTeamRequirements':
      return firebaseSaveRequirements(...args);

    case 'saveIncident':
      return firebaseSaveIncident(...args);

    case 'deleteIncident':
      return firebaseDeleteIncident(...args);

    case 'closeService':
      return firebaseCloseService(...args);

    case 'reopenService':
      return firebaseReopenService(...args);

    case 'addManualWorker':
      return firebaseAddManualWorker(...args);

    case 'replaceWorkersFromPdf':
      return firebaseReplaceWorkersFromPdf(...args);

    case 'generateClosedServicePdf':
      return firebaseGenerateClosedPdf(...args);

    case 'getWorkerHistory': {
      const worker=state.service?.workers?.find(w=>String(w.id)===String(args[1]));
      const relevant=state.service?.incidents?.filter(i=>i.workerId===worker?.id)||[];
      return {movil:worker?.movil||'',services:worker?1:0,present:worker?.status==='PRESENTE'?1:0,
        noShow:worker?.status==='NO PRESENTADO'?1:0,incidents:relevant.length,
        averageRating:'',totalActualMinutes:0,items:[],history:[],workTypes:[]};
    }

    default:
      return callAppsScript(method,args);
  }
}

const backend={
  get run(){
    let successHandler=()=>{};
    let failureHandler=err=>console.error(err);

    let proxy=null;

    proxy=new Proxy({},{
      get(_target,prop){
        if(prop==='withSuccessHandler'){
          return fn=>{
            successHandler=typeof fn==='function'?fn:()=>{};
            return proxy;
          };
        }

        if(prop==='withFailureHandler'){
          return fn=>{
            failureHandler=typeof fn==='function'?fn:failureHandler;
            return proxy;
          };
        }

        if(typeof prop==='symbol')return undefined;

        return (...args)=>{
          Promise.resolve()
            .then(()=>dispatchBackend(String(prop),args))
            .then(result=>successHandler(result))
            .catch(err=>failureHandler(new Error(firebaseErrorMessage(err))));

          return proxy;
        };
      }
    });

    return proxy;
  }
};

let pdfJsLoadingPromise = null;
function ensurePdfJs(){
  if(window.pdfjsLib){
    window.pdfjsLib.GlobalWorkerOptions.workerSrc='https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
    return Promise.resolve(window.pdfjsLib);
  }
  if(pdfJsLoadingPromise)return pdfJsLoadingPromise;

  pdfJsLoadingPromise=new Promise((resolve,reject)=>{
    const script=document.createElement('script');
    script.src='https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.min.js';
    script.async=true;
    script.onload=()=>{
      window.pdfjsLib.GlobalWorkerOptions.workerSrc='https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
      resolve(window.pdfjsLib);
    };
    script.onerror=()=>reject(new Error('No se ha podido cargar el lector de PDF.'));
    document.head.appendChild(script);
  });

  return pdfJsLoadingPromise;
}

function syncPersonnelSidebarPlacement(){
  const sidebar=document.querySelector('.personnel-sidebar-v121');
  const mobileDrawer=document.getElementById('mobileOperationalDrawerV11234');
  if(!sidebar)return;

  const isMobile=window.matchMedia('(max-width: 900px)').matches;

  if(isMobile){
    sidebar.classList.remove('personnel-sidebar-portal-v1215');
    if(mobileDrawer && sidebar.parentElement!==mobileDrawer){
      mobileDrawer.appendChild(sidebar);
    }
  }else{
    setMobileOperationalPanelV11234(false);
    if(sidebar.parentElement!==document.body){
      document.body.appendChild(sidebar);
    }

  }
}

function updateSidebarToggleButton(){
  const btn=document.getElementById('sidebarCollapseBtn');
  const glyph=document.getElementById('sidebarCollapseGlyph');
  const edgeBtn=document.getElementById('sidebarEdgeToggleBtn');
  const edgeGlyph=document.getElementById('sidebarEdgeToggleGlyph');
  const isMobile=window.matchMedia('(max-width: 900px)').matches;
  if(isMobile){
    if(btn)btn.style.display='none';
    if(edgeBtn)edgeBtn.style.display='none';
    document.body.classList.remove('control-sidebar-collapsed');
    return;
  }
  const collapsed=document.body.classList.contains('control-sidebar-collapsed');
  if(btn&&glyph){
    btn.style.display=collapsed?'none':'inline-flex';
    glyph.textContent='‹‹';
    btn.title='Ocultar panel';
    btn.setAttribute('aria-label','Ocultar panel');
  }
  if(edgeBtn&&edgeGlyph){
    edgeBtn.style.display=collapsed?'inline-flex':'none';
    edgeGlyph.textContent='›';
    edgeBtn.title='Mostrar panel';
    edgeBtn.setAttribute('aria-label','Mostrar panel');
  }
}
