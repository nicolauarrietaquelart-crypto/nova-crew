function firebaseErrorMessage(err){
  const raw=String(err?.message||err||'Error de Firebase');
  if(/missing or insufficient permissions/i.test(raw)){
    return 'Firebase está bloqueando el acceso. Revisa Firestore → Rules y permite lectura/escritura durante la prueba.';
  }
  return raw;
}

function firestoreValueToIso(value){
  if(!value)return '';
  if(typeof value==='string')return value;
  if(value&&typeof value.toDate==='function'){
    try{return value.toDate().toISOString()}catch(e){}
  }
  if(value instanceof Date)return value.toISOString();
  return String(value||'');
}

function sanitizeFirestore(value){
  // google.script.run can return array-like/custom Array objects.
  // Firestore only accepts native plain JS Arrays, so Array.from()
  // is used deliberately instead of value.map(), which may preserve
  // the custom Array constructor/species.
  if(Array.isArray(value)){
    return Array.from(value, item=>sanitizeFirestore(item));
  }

  if(value&&typeof value==='object'){
    // Convert special/custom objects into plain Firestore-safe objects.
    if(value instanceof Date){
      return value.toISOString();
    }

    const out={};
    Object.keys(value).forEach(k=>{
      const v=value[k];
      if(v!==undefined)out[k]=sanitizeFirestore(v);
    });
    return out;
  }

  return value;
}

function firebaseServiceRoot(service){
  return sanitizeFirestore({
    code:String(service.code||''),
    serviceName:String(service.serviceName||''),
    serviceDate:String(service.serviceDate||''),
    venue:String(service.venue||''),
    pdfName:String(service.pdfName||''),
    workOrder:Array.from(service.workOrder||[],String),
    hiddenWorkTypes:Array.from(service.hiddenWorkTypes||[],String),
    createdAt:firestoreValueToIso(service.createdAt)||new Date().toISOString(),
    updatedAt:new Date().toISOString(),
    closed:Boolean(service.closed),
    closedAt:firestoreValueToIso(service.closedAt),
    finalPdfId:String(service.finalPdfId||''),
    finalPdfUrl:String(service.finalPdfUrl||''),
    scenarios:Array.from(service.scenarios||[],s=>sanitizeFirestore(s)),
    teams:Array.from(service.teams||[],t=>sanitizeFirestore(t)),
    requirements:Array.from(service.requirements||[],r=>sanitizeFirestore(r)),
    audit:Array.from(service.audit||[],a=>sanitizeFirestore(a)),
    backend:'FIRESTORE',
    schemaVersion:'11.2.34'
  });
}

function normalizeFirebaseWorker(w){
  const worker={...w,id:String(w.id||'')};

  if(typeof calculateClientMinutes==='function'){
    const planned=calculateClientMinutes(worker.inicio,worker.fin,worker.pausa);
    const actual=calculateClientMinutes(
      worker.actualStart||worker.inicio,
      worker.actualEnd||worker.fin,
      worker.actualPause||worker.pausa
    );
    worker.plannedMinutes=planned;
    worker.actualMinutes=actual;
    worker.overtimeMinutes=actual-planned;
  }

  return worker;
}

function normalizeFirebaseIncident(i){
  return {
    ...i,
    id:String(i.id||''),
    createdAt:firestoreValueToIso(i.createdAt)
  };
}

async function commitFirestoreOps(ops){
  if(!ops.length)return;
  const {db,fs}=await ensureFirebase();

  for(let start=0;start<ops.length;start+=400){
    const batch=fs.writeBatch(db);
    ops.slice(start,start+400).forEach(op=>{
      if(op.type==='delete')batch.delete(op.ref);
      else batch.set(op.ref,sanitizeFirestore(op.data),{merge:Boolean(op.merge)});
    });
    await batch.commit();
  }
}

async function replaceFirestoreCollection(code,name,items){
  const {db,fs}=await ensureFirebase();
  const col=fs.collection(db,'services',code,name);
  const snap=await fs.getDocs(col);
  const wanted=new Map(
    (items||[])
      .filter(x=>x&&x.id)
      .map(x=>[String(x.id),sanitizeFirestore(x)])
  );

  const ops=[];
  snap.docs.forEach(d=>{
    if(!wanted.has(d.id))ops.push({type:'delete',ref:d.ref});
  });

  wanted.forEach((data,id)=>{
    ops.push({
      type:'set',
      ref:fs.doc(db,'services',code,name,id),
      data,
      merge:false
    });
  });

  await commitFirestoreOps(ops);
}

async function migrateServiceToFirebase(service){
  if(!service||!service.code)throw new Error('Servicio inválido para Firebase.');

  const {db,fs}=await ensureFirebase();
  const code=String(service.code);

  await fs.setDoc(
    fs.doc(db,'services',code),
    firebaseServiceRoot(service),
    {merge:true}
  );

  await Promise.all([
    replaceFirestoreCollection(
      code,
      'workers',
      (service.workers||[]).map(w=>sanitizeFirestore({...w,id:String(w.id||'')}))
    ),
    replaceFirestoreCollection(
      code,
      'incidents',
      (service.incidents||[]).map(i=>sanitizeFirestore({...i,id:String(i.id||'')}))
    )
  ]);

  return service;
}

async function readServiceFromFirebase(code){
  const {db,fs}=await ensureFirebase();
  const ref=fs.doc(db,'services',code);
  const rootSnap=await fs.getDoc(ref);

  if(!rootSnap.exists())return null;

  const [workersSnap,incidentsSnap]=await Promise.all([
    fs.getDocs(fs.collection(db,'services',code,'workers')),
    fs.getDocs(fs.collection(db,'services',code,'incidents'))
  ]);

  const root=rootSnap.data()||{};
  const workers=workersSnap.docs
    .map(d=>normalizeFirebaseWorker({id:d.id,...d.data()}))
    .sort((a,b)=>Number(a.sortOrder||0)-Number(b.sortOrder||0));

  const incidents=incidentsSnap.docs
    .map(d=>normalizeFirebaseIncident({id:d.id,...d.data()}))
    .sort((a,b)=>String(b.createdAt||'').localeCompare(String(a.createdAt||'')));

  return {
    code:String(code),
    serviceName:String(root.serviceName||''),
    serviceDate:String(root.serviceDate||''),
    venue:String(root.venue||''),
    pdfName:String(root.pdfName||''),
    workOrder:Array.isArray(root.workOrder)?root.workOrder.map(String):[],
    hiddenWorkTypes:Array.isArray(root.hiddenWorkTypes)?root.hiddenWorkTypes.map(String):[],
    createdAt:firestoreValueToIso(root.createdAt),
    updatedAt:firestoreValueToIso(root.updatedAt),
    closed:Boolean(root.closed),
    closedAt:firestoreValueToIso(root.closedAt),
    finalPdfId:String(root.finalPdfId||''),
    finalPdfUrl:String(root.finalPdfUrl||''),
    scenarios:Array.isArray(root.scenarios)?root.scenarios:[],
    teams:Array.isArray(root.teams)?root.teams:[],
    workers,
    incidents,
    requirements:Array.isArray(root.requirements)?root.requirements:[],
    audit:Array.isArray(root.audit)?root.audit:[]
  };
}

function stopFirebaseRealtime(){
  firebaseRealtimeUnsubs.forEach(fn=>{
    try{fn()}catch(e){}
  });
  firebaseRealtimeUnsubs=[];
  firebaseRealtimeCode='';
}

let firebaseOpsRenderTimer=null;
function scheduleOperationalRefresh(){
  clearTimeout(firebaseOpsRenderTimer);
  firebaseOpsRenderTimer=setTimeout(()=>{
    if(!state.service)return;
    renderOperationalAlerts();
  },120);
}

function upsertRealtimeWorker(worker){
  if(!state.service)return null;

  const rows=state.service.workers||[];
  const idx=rows.findIndex(w=>String(w.id)===String(worker.id));
  const previous=idx>=0?rows[idx]:null;

  if(idx>=0)rows[idx]=worker;
  else rows.push(worker);

  rows.sort((a,b)=>Number(a.sortOrder||0)-Number(b.sortOrder||0));
  return previous;
}

function removeRealtimeWorker(id){
  if(!state.service)return null;
  const idx=(state.service.workers||[]).findIndex(w=>String(w.id)===String(id));
  if(idx<0)return null;
  return state.service.workers.splice(idx,1)[0]||null;
}

function refreshWorkerFromRealtime(worker,previous){
  if(!worker||!state.service)return;

  const currentDay=(worker.serviceDate||'SIN FECHA')===state.activeDay;
  const previousDay=previous
    ? (previous.serviceDate||'SIN FECHA')===state.activeDay
    : false;

  // If the worker enters/leaves the active day or work block, rebuild only
  // the personnel area. Normal status/team changes update one row.
  if(!previous || currentDay!==previousDay ||
     String(worker.trabajo||'')!==String(previous.trabajo||'')){
    renderPersonnel(true);
    return;
  }

  if(currentDay){
    refreshWorkerRow(worker.id);

    const currentWork=worker.trabajo||'SIN TIPO';
    const previousWork=previous?.trabajo||currentWork;

    updateGroupSummary(currentWork);
    if(previousWork!==currentWork)updateGroupSummary(previousWork);

    updateStaffingChipFast(currentWork,worker.teamId);
    if(previous?.teamId && previous.teamId!==worker.teamId){
      updateStaffingChipFast(currentWork,previous.teamId);
    }

    // If the interactive team panel is currently open, update it too.
    if(
      !document.getElementById('teamMembersModal')?.classList.contains('hidden') &&
      normalize(state.teamMembersWork)===normalize(currentWork) &&
      (
        String(state.teamMembersTeamId||'')===String(worker.teamId||'') ||
        String(state.teamMembersTeamId||'')===String(previous?.teamId||'')
      )
    ){
      renderTeamMembersModal();
    }
  }

  renderStats();
  scheduleOperationalRefresh();
}

function refreshIncidentWorkerBadge(workerId){
  if(!workerId)return;
  const worker=state.service?.workers?.find(w=>String(w.id)===String(workerId));
  if(worker && (worker.serviceDate||'SIN FECHA')===state.activeDay){
    refreshWorkerRow(worker.id);
  }
}
