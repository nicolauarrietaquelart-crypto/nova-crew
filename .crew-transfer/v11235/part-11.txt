  const scenario=team?state.service.scenarios.find(s=>s.id===team.scenarioId):null;

  switch(key){
    case 'personal': return normalize(w.personal);
    case 'especialidad': return normalize(w.especialidad);
    case 'horario': return clockToMinutes(w.actualStart||w.inicio||'00:00');
    case 'pausa': return clockToMinutes(effectivePause(w)||'00:00');
    case 'dieta': return Number(effectiveDiet(w)||0);
    case 'movil': return normalize(w.movil);
    case 'equipo':
      return normalize(workHasNoTeams(w.trabajo)
        ? 'SIN EQUIPOS'
        : `${scenario?scenario.name:''} ${team?team.name:''}`);
    case 'chaleco': return normalize(team?(team.colorName||team.name):'');
    case 'valoracion': return workerRatingAverage(w);
    case 'estado':
      return {
        'PENDIENTE':1,
        'CONTACTADO':2,
        'PRESENTE':3,
        'NO PRESENTADO':4
      }[w.status||'PENDIENTE']||0;
    default: return Number(w.sortOrder||0);
  }
}

function sortedRowsForWork(trabajo,rows){
  const sort=state.workSorts[trabajo];

  if(!sort||!sort.key){
    return rows.slice().sort((a,b)=>Number(a.sortOrder||0)-Number(b.sortOrder||0));
  }

  const factor=sort.dir==='desc'?-1:1;

  return rows.slice().sort((a,b)=>{
    const av=workerSortValue(a,sort.key);
    const bv=workerSortValue(b,sort.key);

    if(typeof av==='number'&&typeof bv==='number'){
      if(av!==bv)return (av-bv)*factor;
    }else{
      const cmp=String(av??'').localeCompare(String(bv??''),'es',{numeric:true,sensitivity:'base'});
      if(cmp)return cmp*factor;
    }

    return (Number(a.sortOrder||0)-Number(b.sortOrder||0));
  });
}

function filteredRowsForWork(trabajo,rows){
  const filter=state.workTeamFilters[trabajo]||'ALL';

  if(filter==='ALL')return rows;
  if(filter==='UNASSIGNED')return rows.filter(w=>!w.teamId);

  return rows.filter(w=>w.teamId===filter);
}

function staffingSummaryForWork(trabajo,rows){
  if(workHasNoTeams(trabajo)){
    return {
      html:`<div class="staffing-summary">
        <span class="staffing-chip neutral">SIN EQUIPOS · ${rows.length} personas</span>
        <button class="secondary plan-btn" onclick="openStaffingModal('${escapeAttr(trabajo)}')">Configurar</button>
      </div>`,
      readiness:null,
      stats:[]
    };
  }

  const requirements=requirementsForWork(trabajo);
  const teamIds=new Set(requirements.map(r=>r.teamId));

  const counts={};

  rows.forEach(w=>{
    if(!w.teamId)return;

    teamIds.add(w.teamId);

    if(!counts[w.teamId]){
      counts[w.teamId]={assigned:0,present:0};
    }

    counts[w.teamId].assigned++;

    if(w.status==='PRESENTE'){
      counts[w.teamId].present++;
    }
  });

  const requirementMap={};
  requirements.forEach(r=>{
    requirementMap[r.teamId]=Number(r.requiredCount||0);
  });

  const stats=[...teamIds].map(teamId=>{
    const team=state.service.teams.find(t=>t.id===teamId);
    if(!team)return null;

    const count=counts[teamId]||{assigned:0,present:0};

    return {
      team,
      required:Number(requirementMap[teamId]||0),
      assigned:count.assigned,
      present:count.present
    };
  }).filter(Boolean);

  let readiness=null;

  if(requirements.length){
    let requiredTotal=0;
    let assignedCovered=0;
    let presentCovered=0;

    requirements.forEach(r=>{
      const required=Number(r.requiredCount||0);
      const count=counts[r.teamId]||{assigned:0,present:0};

      requiredTotal+=required;
      assignedCovered+=Math.min(required,count.assigned);
      presentCovered+=Math.min(required,count.present);
    });

    if(requiredTotal&&presentCovered>=requiredTotal){
      readiness={cls:'ready',label:'LISTO'};
    }else if(requiredTotal&&assignedCovered>=requiredTotal){
      readiness={cls:'partial',label:'ASIGNADO'};
    }else{
      readiness={cls:'critical',label:'FALTA PERSONAL'};
    }
  }

  let html='';

  if(!requirements.length&&!stats.length){
    html=`<div class="staffing-summary">
      <div class="staffing-empty">
        <span class="hint" style="margin:0">
          Sin equipos asignados ni objetivos definidos. ${rows.length} personas en total.
        </span>
        <button
          class="secondary plan-btn"
          onclick="openStaffingModal('${escapeAttr(trabajo)}')">
          Planificar equipos
        </button>
      </div>
    </div>`;
  }else{
    const chips=stats.map(s=>{
      let cls='neutral';

      if(s.required){
        if(s.present>=s.required)cls='ok';
        else if(s.assigned>=s.required)cls='warn';
        else cls='bad';
      }

      return `<button type="button"
        class="staffing-chip team-clickable ${cls}"
        data-staffing-team="${escapeAttr(s.team.id)}"
        onclick="openTeamMembersModal('${escapeJsString(trabajo)}','${escapeAttr(s.team.id)}')">
        <span class="team-chip-dot" style="background:${escapeAttr(s.team.colorHex||'#bdbdbd')}"></span>
        <strong>${escapeHtml(s.team.name)}</strong>
        <span>${s.assigned}</span>
        <span class="team-chip-present">${s.present} ✓</span>
        ${s.required?`<span class="team-chip-needed">/${s.required}</span>`:''}
      </button>`;
    }).join('');

    html=`<div class="staffing-summary">
      ${chips||'<span class="hint">Sin asignaciones.</span>'}
      <button
        class="secondary plan-btn"
        onclick="openStaffingModal('${escapeAttr(trabajo)}')">
        Planificar
      </button>
    </div>`;
  }

  return {html,readiness,stats};
}

function saveStaffingPlan(){
  if(!state.service||state.service.closed)return;

  const trabajo=state.staffingWork;
  const serviceDate=state.activeDay;

  const noTeams=Boolean(document.getElementById('staffingNoTeams')?.checked);

  const payload=noTeams
    ? []
    : Object.entries(state.staffingDraft)
      .map(([teamId,requiredCount])=>({
        teamId,
        requiredCount:Math.max(0,Math.floor(Number(requiredCount||0)))
      }))
      .filter(x=>x.requiredCount>0);

  // Guardamos una copia solo por si Google Sheets devuelve error.
  const previousRequirements=(state.service.requirements||[]).map(r=>({...r}));

  // ACTUALIZACIÓN LOCAL INSTANTÁNEA:
  // sustituimos los objetivos de este día + tipo de servicio en memoria.
  state.service.requirements=(state.service.requirements||[])
    .filter(r=>!(
      r.serviceDate===serviceDate &&
      normalize(r.trabajo)===normalize(trabajo)
    ))
    .concat(
      noTeams
        ? [{
            serviceDate,
            trabajo,
            teamId:NO_TEAMS_REQUIREMENT_TEAM_ID,
            requiredCount:0
          }]
        : payload.map(r=>({
            serviceDate,
            trabajo,
            teamId:r.teamId,
            requiredCount:r.requiredCount
          }))
    );

  closeStaffingModal();

  // Solo actualiza el resumen afectado; NO vuelve a pintar 158 personas.
  updateGroupSummary(trabajo);
  renderOperationalAlerts();
  setSaveState('Guardando planificación…','saving');

  backend.run
    .withSuccessHandler(result=>{
      // La UI ya estaba actualizada. El servidor solo confirma.
      if(result&&Array.isArray(result.saved)){
        state.service.requirements=(state.service.requirements||[])
          .filter(r=>!(
            r.serviceDate===serviceDate &&
            normalize(r.trabajo)===normalize(trabajo)
          ))
          .concat(result.saved);
      }

      setSaveState('Guardado','saved');
    })
    .withFailureHandler(err=>{
      // Si falla el servidor, restauramos lo anterior para no mentir en pantalla.
      state.service.requirements=previousRequirements;
      updateGroupSummary(trabajo);
      renderOperationalAlerts();
      setSaveState('Error al guardar','error');
      alert(errorMessage(err));
    })
    .saveTeamRequirements(
      state.service.code,
      serviceDate,
      trabajo,
      payload,
      noTeams
    );
}

function scheduleValueHtml(planned,actual){
  const base=escapeHtml(planned||'—');
  if(!actual||actual===planned)return base;
  return `${base}<span class="schedule-real">Real ${escapeHtml(actual)}</span>`;
}


function scheduleRangeHtml(w){
  const plannedStart=w.inicio||'—';
  const plannedEnd=w.fin||'—';
  const planned=`<span class="schedule-planned">${escapeHtml(plannedStart)} – ${escapeHtml(plannedEnd)}</span>`;

  const realStart=w.actualStart||'';
  const realEnd=w.actualEnd||'';
  const hasReal=Boolean(realStart||realEnd);

  if(!hasReal){
    return planned+overnightBadge(w.inicio,w.fin);
  }

  const real=`${escapeHtml(realStart||plannedStart)} – ${escapeHtml(realEnd||plannedEnd)}`;
  return `${planned}<span class="schedule-real">Real ${real}</span>${overnightBadge(realStart||w.inicio,realEnd||w.fin)}`;
}


function hoursSummaryHtml(w){
  const planned=Number(w.plannedMinutes||0);

  // IMPORTANT: 0 real minutes is a valid value.
  // Do not use `w.actualMinutes || planned`, because 0 is falsy.
  const hasActualMinutes=
    w.actualMinutes!==undefined &&
    w.actualMinutes!==null &&
    w.actualMinutes!=='';

  const actual=hasActualMinutes
    ? Number(w.actualMinutes)
    : planned;

  const hasDiff=
    w.overtimeMinutes!==undefined &&
    w.overtimeMinutes!==null &&
    w.overtimeMinutes!=='';

  const diff=hasDiff
    ? Number(w.overtimeMinutes)
    : (actual-planned);

  if(!planned&&!actual)return'';

  let cls='';
  if(diff>0)cls=' extra';
  if(diff<0)cls=' less';

  const text=`${formatMinutesClient(actual)}${diff?` · ${diff>0?'+':''}${formatMinutesClient(diff)}`:''}`;
  return `<span class="hours-chip${cls}">${text}</span>`;
}

function formatMinutesClient(minutes){
  const n=Number(minutes||0);
  const sign=n<0?'-':'';
  const abs=Math.abs(n);
  const h=Math.floor(abs/60);
  const m=abs%60;
  return `${sign}${h}h ${String(m).padStart(2,'0')}m`;
}

function effectivePause(w){
  return w.actualPause!==undefined&&w.actualPause!==null&&w.actualPause!==''
    ? w.actualPause
    : (w.pausa||'00:00');
}

function pauseChipHtml(w){
  const effective=effectivePause(w);
  const manual=w.actualPause!==undefined&&w.actualPause!==null&&w.actualPause!=='';
  const label=effective==='00:00'?'SIN PAUSA':formatPauseLabel(effective);
  return `<span class="ops-chip ${manual?'manual':''}">${escapeHtml(label)}</span>`;
}

function formatPauseLabel(value){
  const m=String(value||'').match(/^(\d{2}):(\d{2})$/);
  if(!m)return value||'—';
  const h=Number(m[1]),min=Number(m[2]);
  if(h===0)return `${min} min`;
  if(min===0)return `${h} h`;
  return `${h} h ${min} min`;
}

function effectiveDiet(w){
  return w.actualDiet!==undefined&&w.actualDiet!==null&&w.actualDiet!==''
    ? Number(w.actualDiet)
    : Number(w.nDieta||0);
}

function dietChipHtml(w){
  const count=effectiveDiet(w);
  const manual=w.actualDiet!==undefined&&w.actualDiet!==null&&w.actualDiet!=='';
  const text=count===0?'SIN DIETA':(count===1?'1 DIETA':`${count} DIETAS`);
  return `<span class="ops-chip ${manual?'manual':''}">${escapeHtml(text)}</span>`;
}

function overnightBadge(start,end){
  if(!isValidClock(start)||!isValidClock(end))return'';

  const startMinutes=clockToMinutes(start);
  const endMinutes=clockToMinutes(end);

  if(endMinutes>=startMinutes)return'';

  return '<span class="schedule-real">+1 día</span>';
}

function isValidClock(value){
  return /^\d{2}:\d{2}$/.test(String(value||''));
}

function clockToMinutes(value){
  const parts=String(value).split(':').map(Number);
  return parts[0]*60+parts[1];
}

function workerRatingAverage(w){
  const vals=[w.ratingUniformidad,w.ratingActitud,w.ratingPuntualidad].map(Number).filter(v=>v>0);
  if(!vals.length)return 0;
  return vals.reduce((a,b)=>a+b,0)/vals.length;
}

function renderStats(){
  if(!state.service)return;
  const workers=state.service.workers.filter(w=>(w.serviceDate||'SIN FECHA')===state.activeDay&&w.included);
  const c={total:workers.length,pendiente:0,contactado:0,presente:0,noPresentado:0};
  workers.forEach(w=>{
    if(w.status==='CONTACTADO')c.contactado++;
    else if(w.status==='PRESENTE')c.presente++;
    else if(w.status==='NO PRESENTADO')c.noPresentado++;
    else c.pendiente++;
  });
  document.getElementById('stats').innerHTML=`
    <div class="stat"><strong>${c.total}</strong><span>Personal del día</span></div>
    <div class="stat"><strong>${c.pendiente}</strong><span>Pendientes</span></div>
    <div class="stat"><strong>${c.contactado}</strong><span>Contactados</span></div>
    <div class="stat"><strong>${c.presente}</strong><span>Presentes</span></div>
    <div class="stat"><strong>${c.noPresentado}</strong><span><span class="sidebar-btn-icon">✕</span>No present.</span></div>`;
}

function refreshShiftSelectionButton(trabajo,time){
  const key=shiftStateKey(trabajo,time);
  const btn=document.querySelector(`[data-shift-select-key="${cssEscape(key)}"]`);
  if(!btn)return;

  const rows=workersInEntryTime(trabajo,time);
  const selected=rows.filter(w=>state.selected.has(w.id)).length;
  const allSelected=rows.length>0 && selected===rows.length;

  btn.classList.toggle('active',allSelected);

  const check=btn.querySelector('.shift-select-check-v1211');
  if(check)check.textContent=allSelected?'✓':'☐';

  const count=btn.querySelector('.shift-selected-count-v121');
  if(count)count.textContent=`${selected}/${rows.length}`;

  btn.title=allSelected
    ? `Quitar de la selección a los ${rows.length} trabajadores de ${time}`
    : `Seleccionar a los ${rows.length} trabajadores de ${time}`;
}

function toggleSelectWorkGroup(trabajo){
  const rows=workersInWorkGroup(trabajo);
  if(!rows.length)return;

  const allSelected=rows.every(w=>state.selected.has(w.id));

  rows.forEach(w=>{
    if(allSelected)state.selected.delete(w.id);
    else state.selected.add(w.id);
  });

  updateGroupSelectionButton(trabajo);
  refreshSelectionRows(rows);
}

function updateGroupSelectionButton(trabajo){
  const rows=workersInWorkGroup(trabajo);
  const allSelected=rows.length>0&&rows.every(w=>state.selected.has(w.id));

  document.querySelectorAll('[data-group-select-all]').forEach(btn=>{
    if(btn.getAttribute('data-group-select-all')!==trabajo)return;

    btn.textContent=(allSelected?'Deseleccionar ':'Seleccionar ')+trabajo;
    btn.classList.toggle('active',allSelected);
  });
}

    state.service.workers
      .filter(w=>(w.serviceDate||'SIN FECHA')===state.activeDay&&w.included)
      .map(w=>w.trabajo||'SIN TIPO')
  )];

  works.forEach(updateGroupSelectionButton);
}

function escapeJsString(value){
  return String(value??'')
    .replace(/\\/g,'\\\\')
    .replace(/'/g,"\\'")
    .replace(/\r?\n/g,' ');
}
