function saveWorkerModal(){
  const w=state.service.workers.find(x=>x.id===state.modalWorkerId);if(!w||!state.modalDraft)return;

  const update={workerId:w.id};

  applyReserveStateLocally(
    w,
    document.getElementById('modalWorkerReserve').checked,
    update
  );

  w.actualStart=document.getElementById('modalActualStart').value||'';
  w.actualEnd=document.getElementById('modalActualEnd').value||'';

  const pauseInput=document.getElementById('modalActualPause');
  w.actualPause=pauseInput.dataset.usePdf==='1'?'':(pauseInput.value||'00:00');

  const dietValue=document.getElementById('modalActualDiet').value;
  w.actualDiet=dietValue==='__PDF__'?null:Number(dietValue);

  w.notes=document.getElementById('modalNotes').value||'';
  w.ratingUniformidad=Number(state.modalDraft.ratingUniformidad||0);
  w.ratingActitud=Number(state.modalDraft.ratingActitud||0);
  w.ratingPuntualidad=Number(state.modalDraft.ratingPuntualidad||0);
  recomputeWorkerMinutes(w);

  Object.assign(update,{
    actualStart:w.actualStart,
    actualEnd:w.actualEnd,
    actualPause:w.actualPause,
    actualDiet:w.actualDiet,
    notes:w.notes,
    ratingUniformidad:w.ratingUniformidad,
    ratingActitud:w.ratingActitud,
    ratingPuntualidad:w.ratingPuntualidad
  });

  saveWorkerUpdates([update]);

  const id=w.id;
  closeWorkerModal();
  refreshWorkerRow(id);
}


function recomputeWorkerMinutes(w){
  const planned=calculateClientMinutes(w.inicio,w.fin,w.pausa||'00:00');
  const actual=calculateClientMinutes(
    w.actualStart||w.inicio,
    w.actualEnd||w.fin,
    (w.actualPause!==undefined&&w.actualPause!==null&&w.actualPause!=='')?w.actualPause:(w.pausa||'00:00')
  );
  w.plannedMinutes=planned;
  w.actualMinutes=actual;
  w.overtimeMinutes=actual-planned;
}

function bestTextColor(hex){
  if(!/^#[0-9a-fA-F]{6}$/.test(hex||''))return'#111';
  const r=parseInt(hex.slice(1,3),16),g=parseInt(hex.slice(3,5),16),b=parseInt(hex.slice(5,7),16);

}

function setupModalCustomScrollbars(){
  document.querySelectorAll('.modal').forEach(modal=>{
    if(modal.dataset.scrollbarReady==='1')return;

    const card=modal.querySelector('.modal-card');
    if(!card)return;

    modal.dataset.scrollbarReady='1';
    card.classList.add('modal-native-scroll-hidden-v11227');

    const rail=document.createElement('div');
    rail.className='modal-custom-scroll-rail-v11227';
    rail.innerHTML='<div class="modal-custom-scroll-thumb-v11227"></div>';
    modal.appendChild(rail);

    const thumb=rail.firstElementChild;

    const update=()=>{
      if(modal.classList.contains('hidden')){
        rail.style.display='none';
        return;
      }

      const max=card.scrollHeight-card.clientHeight;
      if(max<=2){
        rail.style.display='none';
        return;
      }

      const modalRect=modal.getBoundingClientRect();
      const cardRect=card.getBoundingClientRect();
      const inset=8;
      const railHeight=Math.max(40,cardRect.height-inset*2);

      rail.style.display='block';
      rail.style.left=`${Math.max(0,cardRect.right-modalRect.left-9)}px`;
      rail.style.top=`${Math.max(0,cardRect.top-modalRect.top+inset)}px`;
      rail.style.height=`${railHeight}px`;

      const thumbHeight=Math.max(38,railHeight*(card.clientHeight/card.scrollHeight));
      const travel=Math.max(0,railHeight-thumbHeight);
      const pos=max>0?travel*(card.scrollTop/max):0;

      thumb.style.height=`${thumbHeight}px`;
      thumb.style.transform=`translateY(${pos}px)`;
    };

    card.addEventListener('scroll',update,{passive:true});

    rail.addEventListener('pointerdown',event=>{
      if(event.target===thumb)return;
      const rect=rail.getBoundingClientRect();
      const ratio=Math.max(0,Math.min(1,(event.clientY-rect.top)/rect.height));
      card.scrollTop=(card.scrollHeight-card.clientHeight)*ratio;
    });

    thumb.addEventListener('pointerdown',event=>{
      event.preventDefault();
      event.stopPropagation();

      const startY=event.clientY;
      const startScroll=card.scrollTop;
      const max=card.scrollHeight-card.clientHeight;
      const railHeight=rail.clientHeight;
      const thumbHeight=thumb.offsetHeight;
      const travel=Math.max(1,railHeight-thumbHeight);

      thumb.setPointerCapture?.(event.pointerId);

      const move=e=>{
        const delta=e.clientY-startY;
        card.scrollTop=startScroll+(delta/travel)*max;
      };

      const up=()=>{
        window.removeEventListener('pointermove',move);
        window.removeEventListener('pointerup',up);
      };

      window.addEventListener('pointermove',move);
      window.addEventListener('pointerup',up,{once:true});
    });

    const resizeObserver=new ResizeObserver(update);
    resizeObserver.observe(card);

    const mutationObserver=new MutationObserver(()=>requestAnimationFrame(update));
    mutationObserver.observe(modal,{attributes:true,attributeFilter:['class']});

    modal._updateCustomScrollbar=update;
  });
}

function refreshModalCustomScrollbars(){
  setupModalCustomScrollbars();
  requestAnimationFrame(()=>{
    document.querySelectorAll('.modal')
      .forEach(modal=>modal._updateCustomScrollbar?.());
  });
}
