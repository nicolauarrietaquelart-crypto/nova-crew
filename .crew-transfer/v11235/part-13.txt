function updateOperationalWorkNextButton(){
  const host=document.getElementById('operationalWorkFilters');
  const prev=document.getElementById('operationalWorkPrevious');
  const next=document.getElementById('operationalWorkNext');
  if(!host||!prev||!next)return;

  const items=[...host.querySelectorAll('.operational-work-chip-v122')];
  if(!items.length){prev.disabled=true;next.disabled=true;return;}

  const maxIndex=items.length-1;
  const start=Math.max(0,Math.min(maxIndex,operationalWorkPageStartV11233));
  operationalWorkPageStartV11233=start;

  // Measure while temporarily visible; all items fit into a 1-row clipped rail.
  items.forEach(button=>button.hidden=false);
  const widths=items.map(button=>Math.ceil(button.getBoundingClientRect().width));
  const available=Math.max(85,host.clientWidth-4);
  let used=0;
  let last=start;

  for(let index=start;index<items.length;index++){
    const width=Math.min(available,Math.max(36,widths[index]));
    const required=width+(index>start?5:0);
    if(index>start && used+required>available)break;
    used+=required;
    last=index;
  }

  operationalWorkPageLastV11233=last;
  items.forEach((button,index)=>{
    button.hidden=index<start||index>last;
  });

  prev.disabled=start===0;
  next.disabled=last>=maxIndex;
  next.classList.toggle('at-end-v11232',next.disabled);
  host.scrollLeft=0;
}

  const unassigned=workers.filter(w=>
    !w.teamId &&
    !w.isReserve &&

  ).length;

  const noShows=workers.filter(w=>w.status==='NO PRESENTADO').length;
  const reserves=workers.filter(w=>w.isReserve&&w.status!=='NO PRESENTADO').length;

  const workTypes=[...new Set(workers.map(w=>w.trabajo))];

    .filter(r=>
      r.serviceDate===state.activeDay &&

function countScheduleConflicts(workers){
  const byPerson=new Map();

  workers.forEach(w=>{
    const key=w.movil||normalize(w.personal);
    if(!key)return;
    if(!byPerson.has(key))byPerson.set(key,[]);
    byPerson.get(key).push(w);
  });

  let conflicts=0;

  byPerson.forEach(list=>{
    if(list.length<2)return;
    for(let i=0;i<list.length;i++){
      for(let j=i+1;j<list.length;j++){
        if(shiftsOverlap(list[i],list[j]))conflicts++;
      }
    }
  });

  return conflicts;
}

function shiftsOverlap(a,b){
  const aStart=clockMinutes(a.actualStart||a.inicio);
  const aEndRaw=clockMinutes(a.actualEnd||a.fin);
  const bStart=clockMinutes(b.actualStart||b.inicio);
  const bEndRaw=clockMinutes(b.actualEnd||b.fin);

  if(aStart===null||aEndRaw===null||bStart===null||bEndRaw===null)return false;

  let aEnd=aEndRaw,bEnd=bEndRaw;
  if(aEnd<aStart)aEnd+=1440;
  if(bEnd<bStart)bEnd+=1440;

  return (
    Math.max(aStart,bStart)<Math.min(aEnd,bEnd) ||
    Math.max(aStart,bStart+1440)<Math.min(aEnd,bEnd+1440) ||
    Math.max(aStart+1440,bStart)<Math.min(aEnd+1440,bEnd)
  );
}

function buildCloseChecklist(){
  const workers=(state.service.workers||[]).filter(w=>w.included);
  const openIncidents=(state.service.incidents||[]).filter(i=>i.status==='ABIERTA').length;
  const unassigned=workers.filter(w=>!w.teamId&&!w.isReserve&&!workHasNoTeams(w.trabajo)).length;
  const noShows=workers.filter(w=>w.status==='NO PRESENTADO').length;
  const pending=workers.filter(w=>w.status==='PENDIENTE').length;

  let deficits=0;

  (state.service.requirements||[])
    .filter(req=>req.teamId!==NO_TEAMS_REQUIREMENT_TEAM_ID)
    .forEach(req=>{
    const assigned=workers.filter(w=>
      w.serviceDate===req.serviceDate &&
      normalize(w.trabajo)===normalize(req.trabajo) &&
      w.teamId===req.teamId
    ).length;

    deficits+=Math.max(0,Number(req.requiredCount||0)-assigned);
  });

  const warnings=[];
  if(openIncidents)warnings.push(`${openIncidents} incidencia(s) abierta(s)`);
  if(unassigned)warnings.push(`${unassigned} trabajador(es) sin equipo`);
  if(deficits)warnings.push(`${deficits} plaza(s) de planificación sin cubrir`);
  if(pending)warnings.push(`${pending} trabajador(es) todavía pendientes`);
  if(noShows)warnings.push(`${noShows} no presentado(s)`);

  return warnings;
}

function normalizePresentReservesBeforeClose(done){
  const reserves=presentReservesPendingClose();
  if(!reserves.length){
    if(typeof done==='function')done();
    return;
  }

  const updates=[];

  reserves.forEach(w=>{
    const start=w.actualStart||w.inicio||w.arrivalAt||'';
    const end=clockPlusMinutes(start,180);

    w.especialidad='HAND A';
    w.actualStart=start;
    w.actualEnd=end;
    w.actualPause='00:00';
    recomputeWorkerMinutes(w);

    updates.push({
      workerId:w.id,
      especialidad:'HAND A',
      actualStart:start,
      actualEnd:end,
      actualPause:'00:00'
    });
  });

  renderPersonnel();
  renderStats();
  updateGroupSummaries();

  saveWorkerUpdates(updates);
  flushPendingWorkerSaves(()=>{
    if(typeof done==='function')done();
  });
}

function requestCloseService(){
  if(!state.service||state.service.closed)return;

  const warnings=buildCloseChecklist();
  const presentReserves=presentReservesPendingClose();

  let msg=
    `¿Cerrar definitivamente el servicio ${state.service.code}?\n\n`+
    `Se guardarán los cambios y el servicio se bloqueará inmediatamente. El PDF final se generará justo después.`;

  if(warnings.length){
    msg+=`\n\nREVISIÓN ANTES DEL CIERRE:\n• ${warnings.join('\n• ')}`;
  }else{
    msg+='\n\n✓ No se han detectado problemas pendientes.';
  }

  if(presentReserves.length){
    const names=presentReserves.slice(0,6).map(w=>w.personal).join(', ');
    const extra=presentReserves.length>6?` +${presentReserves.length-6} más`:'';

    msg+=
      `\n\nRESERVAS PRESENTES: ${presentReserves.length}`+
      `\nAl cerrar se aplicará automáticamente HAND A + 3 h a quienes sigan marcados como Reserva.`+
      `\n${names}${extra}`+
      `\n\nSi alguna reserva se ha quedado trabajando, cancela y quítale la marca Reserva en su ficha antes de cerrar.`;
  }

  msg+='\n\nPodrás reabrir el servicio posteriormente si fuera necesario.';

  if(!confirm(msg))return;

  setSaveState('Guardando cambios pendientes…','saving');

  flushPendingWorkerSaves(()=>{
    normalizePresentReservesBeforeClose(()=>{
      performCloseService();
    });
  });
}

function performCloseService(){
  if(!state.service||state.service.closed)return;

  const code=state.service.code;
  setSaveState('Cerrando servicio…','saving');

  backend.run
    .withSuccessHandler(result=>{
      // El servicio ya queda cerrado aquí; no esperamos al PDF.
      state.service.closed=true;
      state.service.closedAt=result.closedAt||new Date().toISOString();
      state.service.finalPdfId='';
      state.service.finalPdfUrl=result.pdfUrl||'';

      applyClosedState();
      renderPersonnel();
      renderIncidents();

      if(result.pdfUrl){
        state.service.finalPdfUrl=result.pdfUrl;
        applyClosedState();
        setSaveState('Servicio cerrado','saved');
        return;
      }

      setSaveState('Servicio cerrado · generando PDF…','saving');

      backend.run
        .withSuccessHandler(pdf=>{
          if(pdf?.pdfUrl){
            state.service.finalPdfId=pdf.pdfId||'';
            state.service.finalPdfUrl=pdf.pdfUrl;
            applyClosedState();
          }

          renderAudit();
          setSaveState('Servicio cerrado · PDF listo','saved');
        })
        .withFailureHandler(err=>{
          // El servicio sigue correctamente cerrado aunque falle el PDF.
          setSaveState('Servicio cerrado · PDF pendiente','saved');
          console.error('Error generando PDF final:',errorMessage(err));
        })
        .generateClosedServicePdf(code);
    })
    .withFailureHandler(err=>{
      alert(errorMessage(err));
      setSaveState('Error al cerrar','error');
    })
    .closeService(code);
}

function calculateClientMinutes(start,end,pause){
  const sm=clockMinutes(start),em=clockMinutes(end);
  if(sm===null||em===null)return 0;
  let d=em-sm;if(d<0)d+=1440;
  return Math.max(0,d-(clockMinutes(pause)||0));
}
function clockMinutes(v){
  const m=String(v||'').match(/^(\d{2}):(\d{2})$/);
  return m?Number(m[1])*60+Number(m[2]):null;

function clearBulkRatings(){
  state.bulkRatingDraft={
    ratingUniformidad:null,
    ratingActitud:null,
    ratingPuntualidad:null
  };
  renderBulkRatingStars();
}

function renderBulkRatingStars(){
  renderBulkStarRow('bulkStarsUniformidad','ratingUniformidad',state.bulkRatingDraft.ratingUniformidad);
  renderBulkStarRow('bulkStarsActitud','ratingActitud',state.bulkRatingDraft.ratingActitud);
  renderBulkStarRow('bulkStarsPuntualidad','ratingPuntualidad',state.bulkRatingDraft.ratingPuntualidad);
}

function renderBulkStarRow(containerId,field,value){
  const host=document.getElementById(containerId);
  host.innerHTML=[1,2,3,4,5].map(n=>
    `<button type="button" class="star-btn ${value!==null&&n<=Number(value)?'on':''}" onclick="setBulkRating('${field}',${n})">★</button>`
  ).join('')+
  `<button type="button" class="secondary mini-btn" onclick="setBulkRating('${field}',null)">Sin cambio</button>`;
}

function setBulkRating(field,value){
  state.bulkRatingDraft[field]=value===null?null:Number(value);
  renderBulkRatingStars();
}

function applyBulkRatings(){
  if(!state.selected.size){closeBulkRatingModal();return;}

  const draft=state.bulkRatingDraft||{};
  const hasAny=['ratingUniformidad','ratingActitud','ratingPuntualidad']
    .some(key=>draft[key]!==null&&draft[key]!==undefined);

  if(!hasAny){
    alert('Marca al menos una valoración.');return;
  }

  const updates=[];

  state.service.workers.forEach(w=>{
    if(!state.selected.has(w.id))return;
    const u={workerId:w.id};

    if(draft.ratingUniformidad!==null){
      w.ratingUniformidad=Number(draft.ratingUniformidad);
      u.ratingUniformidad=w.ratingUniformidad;
    }
    if(draft.ratingActitud!==null){
      w.ratingActitud=Number(draft.ratingActitud);
      u.ratingActitud=w.ratingActitud;
    }
    if(draft.ratingPuntualidad!==null){
      w.ratingPuntualidad=Number(draft.ratingPuntualidad);
      u.ratingPuntualidad=w.ratingPuntualidad;
    }

    updates.push(u);
  });

  closeBulkRatingModal();
  renderPersonnel();
  saveWorkerUpdates(updates);
}


function openAddWorkerModal(trabajo){
  if(!state.service||state.service.closed)return;

  state.manualWork=trabajo;

  document.getElementById('addWorkerSub').textContent=
    `${dayLabel(state.activeDay)} · ${state.activeDay} · ${trabajo}`;

  document.getElementById('addWorkerName').value='';
  document.getElementById('addWorkerSpecialty').value='';
  document.getElementById('addWorkerMobile').value='';
  document.getElementById('addWorkerStart').value='';
  document.getElementById('addWorkerEnd').value='';
  document.getElementById('addWorkerPause').value='00:00';
  document.getElementById('addWorkerDiet').value='0';
  document.getElementById('addWorkerReserve').checked=false;
  document.getElementById('addWorkerCoordinator').checked=false;

  const wrap=document.getElementById('addWorkerTeamWrap');
  const select=document.getElementById('addWorkerTeam');

  if(workHasNoTeams(trabajo)){
    wrap.style.display='none';
    select.innerHTML='<option value="">Sin equipos</option>';
  }else{
    wrap.style.display='block';

    let options='<option value="">Sin asignar</option>';

    state.service.scenarios.forEach(s=>{
      const teams=state.service.teams.filter(t=>t.scenarioId===s.id);
      if(!teams.length)return;

      options+=`<optgroup label="${escapeAttr(s.name)}">`;
      teams.forEach(t=>{
        options+=`<option value="${escapeAttr(t.id)}">${escapeHtml(t.name)}</option>`;
      });
      options+='</optgroup>';
    });

    select.innerHTML=options;
  }

  document.getElementById('addWorkerModal').classList.remove('hidden');
}

  state.manualWork='';

function saveManualWorker(){
  if(!state.service||state.service.closed)return;

  const personal=document.getElementById('addWorkerName').value.trim();

  if(!personal){
    alert('Introduce el nombre del trabajador.');
    return;
  }

  const trabajo=state.manualWork;

  if(!confirm(
    `Vas a añadir manualmente a "${personal}" en ${trabajo}.\\n\\n¿Continuar?`
  ))return;

  if(!confirm(
    `SEGUNDA CONFIRMACIÓN\\n\\n¿Seguro que quieres añadir a "${personal}" al servicio?`
  ))return;

  const data={
    serviceDate:state.activeDay,
    trabajo,
    personal,
    especialidad:document.getElementById('addWorkerSpecialty').value.trim(),
    movil:document.getElementById('addWorkerMobile').value.trim(),
    inicio:document.getElementById('addWorkerStart').value,
    fin:document.getElementById('addWorkerEnd').value,
    pausa:document.getElementById('addWorkerPause').value,
    nDieta:Number(document.getElementById('addWorkerDiet').value||0),
    teamId:workHasNoTeams(trabajo)?'':document.getElementById('addWorkerTeam').value,
    isReserve:document.getElementById('addWorkerReserve').checked,
    isCoordinator:document.getElementById('addWorkerCoordinator').checked
  };

  setSaveState('Añadiendo trabajador…','saving');

  backend.run
    .withSuccessHandler(worker=>{
      state.service.workers.push(worker);
      closeAddWorkerModal();
      renderPersonnel();
      renderStats();
      renderOperationalAlerts();
      setSaveState('Trabajador añadido','saved');
    })
    .withFailureHandler(err=>{
      alert(errorMessage(err));
      setSaveState('Error al añadir','error');
    })
    .addManualWorker(state.service.code,data);
}

function deleteCurrentWorker(){
  const w=state.service.workers.find(x=>x.id===state.modalWorkerId);
  if(!w||state.service.closed)return;

  if(!confirm(
    `Vas a eliminar a "${w.personal}" de este servicio.\\n\\n¿Continuar?`
  ))return;

  if(!confirm(
    `SEGUNDA CONFIRMACIÓN\\n\\nEsta persona dejará de aparecer en el servicio. ¿Eliminar a "${w.personal}"?`
  ))return;

  const id=w.id;
  const trabajo=w.trabajo||'SIN TIPO';

  // Soft-delete: queda marcado como excluido en la base.
  // Si se reimporta el PDF, la exclusión se conserva.
  w.included=false;
  state.service.workers=state.service.workers.filter(x=>x.id!==id);
  state.selected.delete(id);

  closeWorkerModal();
  renderPersonnel();
  renderStats();
  updateGroupSummary(trabajo);
  renderOperationalAlerts();

  saveWorkerUpdates([{workerId:id,included:false}]);
}
